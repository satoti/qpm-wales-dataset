# CQ1.1
# Full profile of a place - type, function, geometry, containing units, and its four directional neighbours. Using Saint David's Centre here since I use it as my running example throughout the appendix.
#
# Executed against: QPM_CrossHierarchy_Evaluation_Graph.ttl

import rdflib

g = rdflib.Graph()
g.parse("QPM_CrossHierarchy_Evaluation_Graph.ttl", format="turtle")

query_cq1_1 = r"""
PREFIX qpm: <http://qpm.ontology/2025#>
PREFIX geo: <http://www.opengis.net/ont/geosparql#>

SELECT ?place ?placeName ?place_type ?place_function ?mainGeom
       (GROUP_CONCAT(DISTINCT ?unit_name;  separator=", ") AS ?units)
       (GROUP_CONCAT(DISTINCT ?east_name;  separator=", ") AS ?eastOf)
       (GROUP_CONCAT(DISTINCT ?west_name;  separator=", ") AS ?westOf)
       (GROUP_CONCAT(DISTINCT ?north_name; separator=", ") AS ?northOf)
       (GROUP_CONCAT(DISTINCT ?south_name; separator=", ") AS ?southOf)
WHERE {
  ?place a qpm:Place ;
         qpm:place_name ?placeName ;
         qpm:place_type ?place_type ;
         qpm:place_function ?place_function .
  FILTER(STR(?placeName) = "Saint David's Centre")

  OPTIONAL {
    ?place qpm:hasMainGeometry ?g .
    ?g geo:asWKT ?mainGeom .
  }

  OPTIONAL {
    ?place qpm:contained_by_unit ?u .
    ?u qpm:unit_name ?unit_name .
  }

  OPTIONAL {
    ?place qpm:east_of ?e .
    ?e qpm:place_name ?east_name .
  }

  OPTIONAL {
    ?place qpm:west_of ?w .
    ?w qpm:place_name ?west_name .
  }

  OPTIONAL {
    ?place qpm:north_of ?n .
    ?n qpm:place_name ?north_name .
  }

  OPTIONAL {
    ?place qpm:south_of ?s .
    ?s qpm:place_name ?south_name .
  }
}
GROUP BY ?place ?placeName ?place_type ?place_function ?mainGeom
"""



results = g.query(query_cq1_1)


for row in results:
    print(row)


exit()


# CQ1.2
# Same idea as CQ1.1 but for a unit instead of a place - Castle ward's id, type, level, hierarchy, geometry, what places it contains and its neighbouring units.
#
# Executed against: QPM_CrossHierarchy_Evaluation_Graph.ttl

import rdflib

g = rdflib.Graph()
g.parse("QPM_CrossHierarchy_Evaluation_Graph.ttl", format="turtle")

query_cq1_2 = """
PREFIX qpm: <http://qpm.ontology/2025#>
PREFIX geo: <http://www.opengis.net/ont/geosparql#>

SELECT ?unit ?unitName ?unit_id ?unit_type ?unit_level ?hierarchy 
       ?mainGeom ?extraGeom ?containedBy
       (GROUP_CONCAT(DISTINCT ?placeName; separator=", ") AS ?containsPlaces)
       (GROUP_CONCAT(DISTINCT ?eastName; separator=", ") AS ?eastOf)
       (GROUP_CONCAT(DISTINCT ?westName; separator=", ") AS ?westOf)
       (GROUP_CONCAT(DISTINCT ?northName; separator=", ") AS ?northOf)
       (GROUP_CONCAT(DISTINCT ?southName; separator=", ") AS ?southOf)
WHERE {
  ?unit 
        a qpm:Unit ;
        qpm:unit_name ?unitName ;
        qpm:spatial_unit_id ?unit_id ;
        qpm:unit_type ?unit_type ;
        qpm:unit_level ?unit_level ;
        qpm:belongs_to_hierarchy ?hierarchy .
  FILTER(LCASE(STR(?unitName)) = "castle")

  OPTIONAL {
    ?unit qpm:hasMainGeometry ?mg .
    ?mg geo:asWKT ?mainGeom .
  }
  OPTIONAL {
    ?unit qpm:hasExtraGeometry ?eg .
    ?eg geo:asWKT ?extraGeom .
  }
  OPTIONAL {
    ?unit qpm:contained_by ?parentUnit .
    OPTIONAL { ?parentUnit qpm:unit_name ?parentName.
    }
  }
  BIND(COALESCE(?parentName, "") AS ?containedBy)

  OPTIONAL {
    ?place qpm:contained_by_unit ?unit .
    OPTIONAL { ?place qpm:place_name ?rawPlaceName. }
  }
  BIND(COALESCE(?rawPlaceName, "") AS ?placeName)

  OPTIONAL {
    ?unit qpm:east_of ?e .
    OPTIONAL { ?e qpm:unit_name ?rawEastName. }
  }
  BIND(COALESCE(?rawEastName, "") AS ?eastName)

  OPTIONAL {
    ?unit qpm:west_of ?w .
    OPTIONAL { ?w qpm:unit_name ?rawWestName. }
  }
  BIND(COALESCE(?rawWestName, "") AS ?westName)

  OPTIONAL {
    ?unit qpm:north_of ?n .
    OPTIONAL { ?n qpm:unit_name ?rawNorthName. }
  }
  BIND(COALESCE(?rawNorthName, "") AS ?northName)

  OPTIONAL {
    ?unit qpm:south_of ?s .
    OPTIONAL { ?s qpm:unit_name ?rawSouthName. }
  }
  BIND(COALESCE(?rawSouthName, "") AS ?southName)
}
GROUP BY ?unit ?unitName ?unit_id ?unit_type ?unit_level ?hierarchy 
         ?mainGeom ?extraGeom ?containedBy
"""

results = g.query(query_cq1_2)



count = 0
for row in results:
    print(row)
    count += 1

print(f"Total results: {count}")

exit()

# CQ2.6
# Quick script to double check the betweenness result reported in the
# appendix (Gabalfa -> Castle -> Grangetown). The appendix query uses a
# SPARQL property path over south_of; here I instead do a plain
# breadth-first search over all four directional predicates in Python,
# just to confirm both approaches agree. For this single-hop case they
# do: both return Castle (unit_182) as the unit in between.
#
# Executed against: QPM_CrossHierarchy_Evaluation_Graph.ttl

from rdflib import Graph, Namespace
from collections import deque

QPM = Namespace("http://qpm.ontology/2025#")

g = Graph()
g.parse("QPM_CrossHierarchy_Evaluation_Graph.ttl", format="turtle")

# Define all possible directional predicates
directional_predicates = [
    QPM.south_of,
    QPM.north_of,
    QPM.east_of,
    QPM.west_of
]


def find_units_in_between_auto(graph, start, end, max_length=10):
    """
    Independent validation of CQ2.6 (betweenness inference) via breadth-first
    search across directional predicates, corroborating the SPARQL result
    reported in the thesis Appendix (Listing lst:appendix_between_units).

    Tries each directional predicate in turn and performs a bounded
    breadth-first search (up to max_length hops) from `start` to `end`.
    Returns the intermediate units on the first path found.
    """
    for predicate in directional_predicates:
        queue = deque([(start, [start])])
        visited = set()
        while queue:
            current, path = queue.popleft()
            if current in visited:
                continue
            visited.add(current)
            if current == end:
                print(f"Detected direction: {predicate}")
                return path[1:-1]
            if len(path) > max_length:
                continue
            for neighbor in graph.objects(current, predicate):
                if neighbor not in path:
                    queue.append((neighbor, path + [neighbor]))
    print("No directional path found between the given units.")
    return []


# ===== Example Usage (as reported in the thesis Evaluation Chapter) =====
start_unit = QPM.unit_187  # Gabalfa
end_unit = QPM.unit_188  # Grangetown
between_units = find_units_in_between_auto(g, start_unit, end_unit)
print(f"Units between {start_unit} and {end_unit}:")
if between_units:
    for u in between_units:
        print(" ->", u)
else:
    print("No direct path found.")


# CQ2.7
# Which unit overlaps with Cathays - this is the query behind Appendix Listing lst:appendix_overlapping_units. Expected answer is Cardiff CF103.
#
# Executed against: QPM_CrossHierarchy_Evaluation_Graph.ttl

import rdflib
from rdflib import Graph, Namespace

QPM = Namespace("http://qpm.ontology/2025#")
g = rdflib.Graph()
g.bind("qpm", QPM)
g.parse("QPM_CrossHierarchy_Evaluation_Graph.ttl", format="turtle")

unit_name = "Cathays"

query = f"""
PREFIX qpm: <http://qpm.ontology/2025#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT ?overlapUnitName
WHERE {{
  # find the URI of the given unit
  ?unit a qpm:Unit ;
        qpm:unit_name "{unit_name}"^^xsd:string .

  {{
    # 1) reified pattern
    ?rel a qpm:SpatialRelationship ;
         qpm:sourceUnit   ?unit ;
         qpm:targetUnit   ?overlapUnit ;
         qpm:relationshipType "Overlap"^^xsd:string .
  }}
  UNION
  {{
    # 2) direct pattern
    ?unit qpm:hasSpatialRelationship ?overlapUnit ;
          qpm:relationshipType      "Overlap"^^xsd:string .
  }}

  ?overlapUnit qpm:unit_name ?overlapUnitName .
}}
"""

results = g.query(query)

print(f"Units that overlap with '{unit_name}':")
found = False
for row in results:
    print(f"- {row.overlapUnitName}")
    found = True

if not found:
    print(f"No overlapping units found for '{unit_name}'.")

# CQ5.2
# Provenance details for each computed place - what relationship type and which anchor place(s) it was derived from.
#
# Executed against: QPM_CrossHierarchy_Evaluation_Graph.ttl

import rdflib

g = rdflib.Graph()
g.parse("QPM_CrossHierarchy_Evaluation_Graph.ttl", format="turtle")

query_cq5_2 = """
PREFIX qpm: <http://qpm.ontology/2025#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT ?place ?place_name ?relationship ?involved_place ?involved_place_name
WHERE {
    ?place a qpm:Place ;
           qpm:geometry_source "Computed"^^xsd:string ;
           qpm:place_name ?place_name ;
           qpm:hasProvenance ?provenance .
    ?provenance qpm:involved_place ?involved_place .
    ?involved_place qpm:place_name ?involved_place_name .
    OPTIONAL { ?provenance qpm:described_relationship ?relationship . }
}
"""

# Execute the query
results = g.query(query_cq5_2)

# Print the results
print(f"{'Computed Place':<20} | {'Relationship':<15} | {'Involved Place'}")
print("-" * 70)

count = 0
for row in results:
    place_name = str(row.place_name) if row.place_name else "N/A"
    relationship = str(row.relationship) if row.relationship else "N/A"
    involved_name = str(row.involved_place_name) if row.involved_place_name else "N/A"

    print(f"{place_name:<20} | {relationship:<15} | {involved_name}")
    count += 1

print("-" * 70)
print(f"Total provenance records: {count}")
